# Image CDN Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Serve listing and service photos through a Cloudflare Worker on `img.hasio.net` that
edge-caches them and converts them to AVIF/WebP, without shipping any mobile app update.

**Architecture:** A new, standalone Worker proxies `img.hasio.net/i/<storageId>` to Convex file
storage with `cf.image` transformations and a one-year immutable cache. The Convex `getStorageUrl`
query — whose only two callers are upload paths in the already-shipped clients — is changed to
return the CDN form, so new uploads are routed without touching either client. A one-time,
reversible internal mutation rewrites the URLs already stored in `listings.images` and
`services.images`.

**Tech Stack:** Cloudflare Workers (wrangler 4, already a devDependency), Cloudflare Image
Transformations, Convex, Vitest with `edge-runtime` + `convex-test`.

**Spec:** `docs/superpowers/specs/2026-09-12-image-cdn-design.md`

```ts
/**
 * Mapping between Convex file-storage URLs and their `img.hasio.net` CDN form.
 *
 * Pure on purpose. `getStorageUrl` goes through `getAuthenticatedAppUser`, which
 * convex-test cannot resolve (see `convex/test.utils.ts`), so the query itself is
 * untestable in this suite. Keeping every decision here means the part that can
 * carry a bug is the part that is covered.
 *
 * Every function takes its configuration as an argument rather than reading the
 * environment directly, so tests never depend on deployment state.
 */

/** 8-4-4-4-12 lowercase hex. Verified against all 41 ids in production, 2026-09-12. */
const STORAGE_ID = "[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}";

/**
 * A Convex storage URL. The host allows one or two labels before `convex.cloud`
 * because production is regional (`hearty-ram-74.eu-west-1.convex.cloud`) while
 * dev deployments are not. Anchored at both ends so a lookalike host such as
 * `convex.cloud.evil.com` cannot match.
 */
const CONVEX_STORAGE_URL = new RegExp(
  `^https://[a-z0-9-]+(?:\\.[a-z0-9-]+)?\\.convex\\.cloud/api/storage/(${STORAGE_ID})$`,
  "i"
);

const CDN_PATH = "/i/";

/** Strip trailing slashes so callers can pass either form. */
function trimSlash(base: string): string {
  return base.replace(/\/+$/, "");
}

/** The storage id inside a Convex storage URL, or null if it is not one. */
export function storageIdFromConvexUrl(url: string): string | null {
  const match = CONVEX_STORAGE_URL.exec(url);
  return match ? match[1].toLowerCase() : null;
}

/** The storage id inside a CDN URL, or null if it is not one. */
export function storageIdFromCdnUrl(url: string, base: string): string | null {
  const prefix = `${trimSlash(base)}${CDN_PATH}`;
  if (!url.startsWith(prefix)) return null;
  const id = url.slice(prefix.length).split("?")[0];
  return new RegExp(`^${STORAGE_ID}$`, "i").test(id) ? id.toLowerCase() : null;
}

/**
 * Convex storage URL → CDN URL.
 *
 * Anything else is returned untouched: Unsplash seed URLs, URLs already on the
 * CDN, and — critically — every URL when `base` is null. That last case is what
 * makes it safe to deploy this before the Worker exists, and what makes a dev
 * deployment with no `IMAGE_CDN_URL` behave exactly as it does today.
 */
export function toCdnUrl(url: string | null, base: string | null): string | null {
  if (!url || !base) return url;
  const id = storageIdFromConvexUrl(url);
  return id ? `${trimSlash(base)}${CDN_PATH}${id}` : url;
}

/**
 * CDN URL → Convex storage URL. The exact inverse of `toCdnUrl`, which is what
 * makes the one-time rewrite reversible: the storage id survives in the CDN URL,
 * so nothing is lost and the rollback restores the original strings byte for byte.
 */
export function toConvexUrl(url: string, convexOrigin: string, base: string | null): string {
  if (!base) return url;
  const id = storageIdFromCdnUrl(url, base);
  return id ? `${trimSlash(convexOrigin)}/api/storage/${id}` : url;
}

/**
 * The configured CDN base, or null when unset.
 *
 * Null is a supported state, not an error: it means "behave exactly as before".
 * The migration in `convex/listings/imageCdn.ts` is the one caller that refuses
 * to proceed without it, because silently rewriting nothing would look like success.
 */
export function cdnBase(): string | null {
  const base = process.env.IMAGE_CDN_URL;
  return base ? trimSlash(base) : null;
}
```

```ts
/**
 * Request parsing for the image Worker. Pure, so it is testable without a
 * Worker runtime — `index.ts` stays thin enough to read in one screen.
 */

/**
 * The widths a client may ask for. An allowlist rather than a free number for
 * two reasons: Cloudflare bills per unique transformation, and each distinct
 * width is a separate cache entry, so an unbounded `?w=` leaks both money and
 * cache hit rate.
 *
 * Must stay sorted ascending — `snapWidth` relies on it.
 */
export const ALLOWED_WIDTHS = [200, 400, 600, 800, 1000, 1400] as const;

/**
 * What a request with no `?w=` gets. Sized for the largest real use (the detail
 * sheet's gallery) because the shipped binaries cannot send a width — they ask
 * for one URL and use it everywhere.
 */
export const DEFAULT_WIDTH = 1000;

/** 8-4-4-4-12 hex. Verified against all 41 storage ids in production, 2026-09-12. */
const STORAGE_ID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

/** Round a requested width up to the next allowed one, capped at the largest. */
export function snapWidth(raw: string | null): number {
  if (!raw) return DEFAULT_WIDTH;
  const n = Number(raw);
  if (!Number.isFinite(n) || n <= 0) return DEFAULT_WIDTH;
  return ALLOWED_WIDTHS.find((w) => w >= n) ?? ALLOWED_WIDTHS[ALLOWED_WIDTHS.length - 1];
}

export type ImageRequest = { storageId: string; width: number };

/**
 * `/i/<storageId>` → the id and width to serve, or null for anything else.
 *
 * The id pattern is the security boundary: the Worker builds an origin URL from
 * whatever it captures here, so only a literal storage id may get through. A
 * path segment containing a scheme, a slash or traversal fails the match and
 * 404s, which is what keeps this from being a proxy for arbitrary origins.
 */
export function parseImageRequest(url: URL): ImageRequest | null {
  const match = /^\/i\/([^/]+)$/.exec(url.pathname);
  if (!match) return null;

  const storageId = decodeURIComponent(match[1]);
  if (!STORAGE_ID.test(storageId)) return null;

  return { storageId: storageId.toLowerCase(), width: snapWidth(url.searchParams.get("w")) };
}
```

```ts
import { parseImageRequest } from "./url";

export interface Env {
  /** Convex deployment origin, e.g. https://hearty-ram-74.eu-west-1.convex.cloud */
  CONVEX_ORIGIN: string;
}

/** One year. Safe because a Convex storage id is immutable — it can never come
 *  to point at different bytes, so a cached copy can never go stale. */
const IMMUTABLE = "public, max-age=31536000, immutable";

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    if (request.method !== "GET" && request.method !== "HEAD") {
      return new Response("Method not allowed", {
        status: 405,
        headers: { allow: "GET, HEAD" },
      });
    }

    const parsed = parseImageRequest(new URL(request.url));
    if (!parsed) {
      return new Response("Not found", { status: 404 });
    }

    const origin = `${env.CONVEX_ORIGIN}/api/storage/${parsed.storageId}`;

    const upstream = await fetch(origin, {
      // Forward Accept so `format: "auto"` can negotiate AVIF/WebP per device.
      headers: { accept: request.headers.get("accept") ?? "image/*" },
      cf: {
        image: {
          width: parsed.width,
          format: "auto",
          fit: "scale-down", // never upscale: a 736px source asked for at 1000 stays 736
          quality: 78,
          // If the transformation fails for any reason — including a 9422 when
          // the free-tier transform quota is exhausted — serve the untransformed
          // original rather than an error. A broken pipeline degrades to today's
          // behaviour, not to a broken image.
          onerror: "redirect",
        },
        cacheEverything: true,
        cacheTtl: 31536000,
      },
    } as RequestInit);

    if (!upstream.ok) {
      return new Response("Upstream error", { status: upstream.status === 404 ? 404 : 502 });
    }

    const response = new Response(upstream.body, upstream);
    response.headers.set("cache-control", IMMUTABLE);
    // Convex answers `private`; the stored copy must not inherit that or no edge
    // will ever hold it, which is the whole problem this Worker exists to fix.
    response.headers.delete("set-cookie");
    response.headers.delete("vary");
    response.headers.set(
      "x-hasio-img",
      `${parsed.width}/${response.headers.get("content-type") ?? "unknown"}`
    );
    return response;
  },
};
```

```ts
import { internalMutation, internalQuery } from "../_generated/server";
import { v } from "convex/values";
import { cdnBase, toCdnUrl, toConvexUrl } from "../lib/imageUrl";

/**
 * One-time migration of already-stored photo URLs onto the image CDN, and its
 * exact inverse.
 *
 * Run from the CLI:
 *
 *   npx convex run listings/imageCdn:previewRewrite --prod     # dry run, writes nothing
 *   npx convex run listings/imageCdn:rewriteToCdn --prod       # the migration
 *   npx convex run listings/imageCdn:revertToConvex --prod     # the rollback
 *
 * Internal, so no client can reach them — only someone who already holds deploy
 * credentials, matching the pattern in `convex/admin/devTools.ts`.
 *
 * The rollback is exact because the storage id survives inside the CDN URL, so
 * nothing is lost in the mapping. What it cannot undo is a binary that has
 * already cached a CDN URL — `img.hasio.net` has to keep answering regardless.
 */

/** How many rows a single run will touch. Far above the ~56 listings in
 *  production, low enough that the migration can never run away. */
const MAX_ROWS = 2000;

/**
 * `listings` and `services` are the only tables holding public photo URL arrays,
 * and each is walked explicitly rather than through a loop over table names.
 *
 * That looks like duplication, and it is — but `ctx.db.patch` resolves its value
 * type from the table its id belongs to, so a `"listings" | "services"` union
 * never narrows and the whole thing fails to typecheck. The shared part that
 * could actually carry a bug is `rewriteImageList` / `revertImageList` above,
 * which is pure and covered. What is repeated here is eight mechanical lines.
 *
 * `users.cvFileId` is deliberately absent: business documents are private and
 * must never be proxied.
 */

export function rewriteImageList(
  images: string[],
  base: string
): { images: string[]; changed: number } {
  let changed = 0;
  const next = images.map((url) => {
    const mapped = toCdnUrl(url, base)!;
    if (mapped !== url) changed++;
    return mapped;
  });
  return { images: next, changed };
}

export function revertImageList(
  images: string[],
  convexOrigin: string,
  base: string
): { images: string[]; changed: number } {
  let changed = 0;
  const next = images.map((url) => {
    const mapped = toConvexUrl(url, convexOrigin, base);
    if (mapped !== url) changed++;
    return mapped;
  });
  return { images: next, changed };
}

/** The Convex deployment origin, for rebuilding storage URLs on revert. */
function convexOrigin(override?: string): string {
  const origin = override ?? process.env.CONVEX_CLOUD_URL;
  if (!origin) {
    throw new Error(
      "CONVEX_CLOUD_URL is not set. Pass the origin explicitly: " +
        `npx convex run listings/imageCdn:revertToConvex '{"convexOrigin":"https://….convex.cloud"}' --prod`
    );
  }
  return origin;
}

/** The configured base, or a loud failure. A migration that silently rewrites
 *  nothing looks exactly like a migration that succeeded. */
function requireBase(): string {
  const base = cdnBase();
  if (!base) {
    throw new Error(
      "IMAGE_CDN_URL is not set on this deployment. Set it in the Convex dashboard " +
        "(Settings → Environment Variables) to https://img.hasio.net before migrating."
    );
  }
  return base;
}

/** Dry run: how many rows and URLs would change, writing nothing. */
export const previewRewrite = internalQuery({
  args: {},
  handler: async (ctx) => {
    const base = requireBase();

    const count = (rows: { images?: string[] }[]) => {
      let touchedRows = 0;
      let touchedUrls = 0;
      for (const row of rows) {
        if (!row.images?.length) continue;
        const { changed } = rewriteImageList(row.images, base);
        if (changed > 0) {
          touchedRows++;
          touchedUrls += changed;
        }
      }
      return { rows: touchedRows, urls: touchedUrls, scanned: rows.length };
    };

    return {
      base,
      listings: count(await ctx.db.query("listings").take(MAX_ROWS)),
      services: count(await ctx.db.query("services").take(MAX_ROWS)),
    };
  },
});

/** Rewrite stored Convex storage URLs onto the CDN. Idempotent. */
export const rewriteToCdn = internalMutation({
  args: {},
  handler: async (ctx) => {
    const base = requireBase();
    const now = Date.now();
    const listings = { rows: 0, urls: 0 };
    const services = { rows: 0, urls: 0 };

    for (const row of await ctx.db.query("listings").take(MAX_ROWS)) {
      if (!row.images?.length) continue;
      const { images, changed } = rewriteImageList(row.images, base);
      if (changed === 0) continue;
      await ctx.db.patch(row._id, { images, updatedAt: now });
      listings.rows++;
      listings.urls += changed;
    }

    for (const row of await ctx.db.query("services").take(MAX_ROWS)) {
      if (!row.images?.length) continue;
      const { images, changed } = rewriteImageList(row.images, base);
      if (changed === 0) continue;
      await ctx.db.patch(row._id, { images, updatedAt: now });
      services.rows++;
      services.urls += changed;
    }

    return { base, listings, services };
  },
});

/** The rollback. Exact, because the storage id survives in the CDN URL. */
export const revertToConvex = internalMutation({
  args: { convexOrigin: v.optional(v.string()) },
  handler: async (ctx, args) => {
    const base = requireBase();
    const origin = convexOrigin(args.convexOrigin);
    const now = Date.now();
    const listings = { rows: 0, urls: 0 };
    const services = { rows: 0, urls: 0 };

    for (const row of await ctx.db.query("listings").take(MAX_ROWS)) {
      if (!row.images?.length) continue;
      const { images, changed } = revertImageList(row.images, origin, base);
      if (changed === 0) continue;
      await ctx.db.patch(row._id, { images, updatedAt: now });
      listings.rows++;
      listings.urls += changed;
    }

    for (const row of await ctx.db.query("services").take(MAX_ROWS)) {
      if (!row.images?.length) continue;
      const { images, changed } = revertImageList(row.images, origin, base);
      if (changed === 0) continue;
      await ctx.db.patch(row._id, { images, updatedAt: now });
      services.rows++;
      services.urls += changed;
    }

    return { origin, listings, services };
  },
});
```

```markdown
normally. `getBusinessDocUrl` is deliberately excluded: business licences and ID documents are
```
